```python
def parse_avito(
    query: str,
    max_price: int,
    city: str = "msk",
    limit: int = 10
) -> list[dict]:
    """
    Парсит лоты с Avito по запросу.

    Args:
        query: что ищем ("iPhone 13 128GB")
        max_price: верхняя граница цены в рублях
        city: код города — msk / spb / krd / all
        limit: максимум результатов (default 10, max 50)

    Returns:
        [
            {
                "id": "avito_xxx",
                "title": "iPhone 13 128GB",
                "price": 24000,
                "url": "https://www.avito.ru/...",
                "photos": ["https://..."],
                "seller": "Иван",
                "city": "Москва",
                "posted": "2026-06-08"  # ISO date
            },
            ...
        ]

    Raises:
        BlockedError: VPS IP забанен (такие уже были!)
        ParseError: Avito сменил вёрстку
        AuthError: куки протухли
    """
```

## ⚙️ ТЕХНИЧЕСКИЕ ТРЕБОВАНИЯ

**Стек:**
- Python 3.12
- Selenium 4 + Chrome (есть в системе, проверь `chrome --version`)
- BeautifulSoup4 для парсинга HTML
- fake-useragent для ротации
- requests как fallback (быстрее, но чаще банят)

**Логика:**
1. Открыть `https://www.avito.ru/{city}?q={query}&pmin=...&pmax={max_price}`
2. Подождать загрузку 3-5 сек (Selenium)
3. Спарсить карточки лотов
4. Извлечь: title, price, url, photos, seller, city, posted
5. Отфильтровать по max_price (если Avito пропустил лишнее)
6. Вернуть top `limit`

**Защита от бана (КРИТИЧНО!):**
- User-Agent рандомный из fake-useragent
- Задержка между запросами 5-10 сек (не 1 сек!)
- Если 429 / капча / "Доступ ограничен" → raise `BlockedError`
- Retry: 2 попытки, пауза 30 сек между
- НЕ делать больше 30 запросов в час

**Сортировка:** дешевые сначала (по умолчанию). Опционально — свежие (по дате).

```python
def test_basic_search():
    """Поиск iPhone 13 128GB до 35000 в Москве"""
    results = parse_avito("iPhone 13 128GB", 35000, "msk", limit=5)
    assert len(results) >= 1
    assert all(r["price"] <= 35000 for r in results)
    assert all("iphone" in r["title"].lower() for r in results)
    print(f"OK: {len(results)} результатов, диапазон цен: {min(r['price'] for r in results)} - {max(r['price'] for r in results)}₽")

def test_no_results():
    """Запрос с заведомо низкой ценой"""
    results = parse_avito("iPhone 15 Pro Max 1TB", 100, "msk", limit=5)
    assert results == []
```
